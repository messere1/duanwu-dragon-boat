cmake_minimum_required(VERSION 3.15)
project(DragonBoat LANGUAGES CXX)
if(NOT WIN32)
 message(FATAL_ERROR "This renderer uses Win32/WGL and Windows Imaging Component.")
endif()
add_executable(DragonBoat WIN32 src/main.cpp)
target_compile_features(DragonBoat PRIVATE cxx_std_17)
target_compile_definitions(DragonBoat PRIVATE UNICODE _UNICODE)
target_link_libraries(DragonBoat PRIVATE opengl32 gdi32 comdlg32 shell32 user32 ole32 windowscodecs uuid)
set_target_properties(DragonBoat PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
 set_target_properties(DragonBoat PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_CURRENT_SOURCE_DIR}/bin")
endforeach()
if(MSVC)
 target_compile_options(DragonBoat PRIVATE /utf-8 /W4)
else()
 target_link_options(DragonBoat PRIVATE -municode)
 if(CMAKE_CXX_COMPILER_VERSION VERSION_LESS 8)
  target_link_libraries(DragonBoat PRIVATE stdc++fs)
 endif()
endif()
